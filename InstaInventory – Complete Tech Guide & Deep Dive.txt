3.2 Tables

shops

Stores each seller's shop. One shop per auth user (1-to-1).

CREATE TABLE public.shops (
  id            UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  owner_id      UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  name          TEXT NOT NULL,
  slug          TEXT NOT NULL UNIQUE,               -- URL-friendly shop identifier
  logo_url      TEXT,                                -- Supabase Storage URL
  whatsapp      TEXT NOT NULL,                       -- e.g. "+201001234567"
  instapay_name TEXT,                                -- InstaPay account holder name
  instapay_number TEXT,                              -- InstaPay number (e.g. "01001234567")
  currency      TEXT NOT NULL DEFAULT 'EGP',
  default_shipping_fee  NUMERIC(10,2) NOT NULL DEFAULT 50.00,
  order_counter INT NOT NULL DEFAULT 0,              -- Per-shop order number counter
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at    TIMESTAMPTZ NOT NULL DEFAULT now(),

  CONSTRAINT shops_owner_unique UNIQUE (owner_id)    -- one shop per user
);

CREATE INDEX idx_shops_owner ON public.shops(owner_id);
CREATE INDEX idx_shops_slug  ON public.shops(slug);

3.3 Database Functions & Triggers

Auto-update updated_at

CREATE OR REPLACE FUNCTION public.handle_updated_at()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = now();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Apply to all relevant tables
CREATE TRIGGER set_updated_at BEFORE UPDATE ON public.shops
  FOR EACH ROW EXECUTE FUNCTION public.handle_updated_at();

CREATE TRIGGER set_updated_at BEFORE UPDATE ON public.products
  FOR EACH ROW EXECUTE FUNCTION public.handle_updated_at();

CREATE TRIGGER set_updated_at BEFORE UPDATE ON public.product_variants
  FOR EACH ROW EXECUTE FUNCTION public.handle_updated_at();

CREATE TRIGGER set_updated_at BEFORE UPDATE ON public.customers
  FOR EACH ROW EXECUTE FUNCTION public.handle_updated_at();

CREATE TRIGGER set_updated_at BEFORE UPDATE ON public.orders
  FOR EACH ROW EXECUTE FUNCTION public.handle_updated_at();

Per-Shop Order Number Generation

-- Atomically increment shop counter and return next order number
CREATE OR REPLACE FUNCTION public.next_order_number(p_shop_id UUID)
RETURNS INT AS $$
DECLARE
  v_next INT;
BEGIN
  UPDATE public.shops
  SET order_counter = order_counter + 1
  WHERE id = p_shop_id
  RETURNING order_counter INTO v_next;

  RETURN v_next;
END;
$$ LANGUAGE plpgsql;

-- Auto-assign order_number on insert
CREATE OR REPLACE FUNCTION public.assign_order_number()
RETURNS TRIGGER AS $$
BEGIN
  NEW.order_number = public.next_order_number(NEW.shop_id);
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER set_order_number
  BEFORE INSERT ON public.orders
  FOR EACH ROW
  EXECUTE FUNCTION public.assign_order_number();

Stock Deduction on Pending (Reservation) & Status Tracking

-- ★ STOCK DEDUCTED ON 'pending' (when checkout link is generated)
-- ★ STOCK RESTORED ON 'cancelled' (from pending or confirmed)
CREATE OR REPLACE FUNCTION public.handle_order_status_change()
RETURNS TRIGGER AS $$
BEGIN
  -- Deduct stock when order moves to 'pending' (checkout link sent)
  IF OLD.status = 'draft' AND NEW.status = 'pending' THEN
    NEW.pending_at = now();
    UPDATE public.product_variants pv
    SET stock_qty = stock_qty - oi.quantity
    FROM public.order_items oi
    WHERE oi.order_id = NEW.id
      AND pv.id = oi.variant_id;
  END IF;

  -- Restore stock on cancellation (from pending or confirmed)
  IF OLD.status IN ('pending', 'confirmed') AND NEW.status = 'cancelled' THEN
    NEW.cancelled_at = now();
    UPDATE public.product_variants pv
    SET stock_qty = stock_qty + oi.quantity
    FROM public.order_items oi
    WHERE oi.order_id = NEW.id
      AND pv.id = oi.variant_id;
  END IF;

  -- Track status timestamps
  IF NEW.status = 'confirmed' AND OLD.status != 'confirmed' THEN
    NEW.confirmed_at = now();
  END IF;

  IF NEW.status = 'shipped' AND OLD.status != 'shipped' THEN
    NEW.shipped_at = now();
  END IF;

  IF NEW.status = 'delivered' AND OLD.status != 'delivered' THEN
    NEW.delivered_at = now();
  END IF;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER on_order_status_change
  BEFORE UPDATE OF status ON public.orders
  FOR EACH ROW
  EXECUTE FUNCTION public.handle_order_status_change();

Recalculate Order Totals

-- Recalculate subtotal/total after order_items change
CREATE OR REPLACE FUNCTION public.recalculate_order_total()
RETURNS TRIGGER AS $$
DECLARE
  v_subtotal NUMERIC(10,2);
  v_shipping NUMERIC(10,2);
BEGIN
  SELECT COALESCE(SUM(line_total), 0) INTO v_subtotal
  FROM public.order_items
  WHERE order_id = COALESCE(NEW.order_id, OLD.order_id);

  SELECT shipping_fee INTO v_shipping
  FROM public.orders
  WHERE id = COALESCE(NEW.order_id, OLD.order_id);

  UPDATE public.orders
  SET subtotal = v_subtotal,
      total = v_subtotal + v_shipping
  WHERE id = COALESCE(NEW.order_id, OLD.order_id);

  RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER recalc_order_total_insert
  AFTER INSERT ON public.order_items
  FOR EACH ROW EXECUTE FUNCTION public.recalculate_order_total();

CREATE TRIGGER recalc_order_total_update
  AFTER UPDATE ON public.order_items
  FOR EACH ROW EXECUTE FUNCTION public.recalculate_order_total();

CREATE TRIGGER recalc_order_total_delete
  AFTER DELETE ON public.order_items
  FOR EACH ROW EXECUTE FUNCTION public.recalculate_order_total();

3.4 Row Level Security (RLS)

Principle: Every seller sees only their own shop data. Public checkout uses the checkout_token — no auth required.



-- Enable RLS on all tables
ALTER TABLE public.shops            ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.products         ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.product_images   ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.product_variants ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.customers        ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.orders           ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.order_items      ENABLE ROW LEVEL SECURITY;

Shops

-- Sellers can only access their own shop
CREATE POLICY "Sellers manage own shop"
  ON public.shops FOR ALL
  USING (owner_id = auth.uid())
  WITH CHECK (owner_id = auth.uid());

Products

CREATE POLICY "Sellers manage own products"
  ON public.products FOR ALL

5. Payment Flow Design

5.1 Two Payment Methods

Payment Method Selection (checkout page)
├── COD (الدفع عند الاستلام)
│   └── Just select → submit order → done
│
└── InstaPay (إنستاباي)
    ├── Show seller's InstaPay details (from shop settings)
    ├── Buyer transfers money externally in InstaPay app
    ├── Buyer returns to checkout → uploads screenshot
    └── Submit order with screenshot → seller reviews in dashboard

5.2 Screenshot Upload Flow

Buyer selects "InstaPay"
    ↓
UI shows: "Transfer to [shop InstaPay number]. Then upload screenshot below."
    ↓
Buyer taps upload → selects image from phone gallery
    ↓
Client-side: validate file (jpeg/png/webp, <10MB)
    ↓
POST /api/upload → Supabase Storage "payment-proofs/{shop_id}/{order_id}.jpg"
    ↓
Returns signed URL → stored in orders.payment_proof_url
    ↓
Seller sees screenshot thumbnail in order detail page
    ↓
Seller manually verifies → marks order "Confirmed"

5.3 Key Design Decisions

No auto-verification: Seller visually checks the screenshot and confirms manually
Storage privacy: Payment proofs are in a private bucket — only the shop owner can view via signed URL
No external gateway: Per spec — "No external payment integration for now"

10. Checkout Token Security

Token design:
• Generated via nanoid(21) → URL-safe, 21 chars → ~124 bits of entropy
• Stored in orders.checkout_token (UNIQUE index)
• Public URL format: /checkout/{token}
• Token is the ONLY way to access order data publicly
• No rate limiting needed in V1 (brute force infeasible at 124 bits)


11. Order State Machine

                  ┌──────────┐
                  │  draft   │  Seller building cart (no stock impact)
                  └────┬─────┘
                       │ Generate checkout link
                       ▼
                  ┌──────────┐
         ┌───────│ pending  │  ★ STOCK DEDUCTED (reserved for buyer)
         │       └────┬─────┘
         │            │ Seller confirms (after buyer submits or InstaPay verified)
         │            ▼
         │       ┌──────────┐
         │  ┌───│confirmed │  Order locked in
         │  │   └────┬─────┘
         │  │        │ Seller marks shipped
         │  │        ▼
         │  │   ┌──────────┐
         │  │   │ shipped  │
         │  │   └────┬─────┘
         │  │        │ Seller marks delivered
         │  │        ▼
         │  │   ┌──────────┐
         │  │   │delivered │  ✅ Complete
         │  │   └──────────┘
         │  │
         │  │ (from pending OR confirmed)
         ▼  ▼
    ┌──────────┐
    │cancelled │  ★ STOCK RESTORED
    └──────────┘

Dashboard hint: Orders in pending for >2 hours show a "⏰ في الانتظار منذ ساعتين" badge so the seller remembers to follow up or cancel stale orders to release stock.



Allowed transitions:

From To Who Stock Effect
draft pending Seller (generates link) Deducted
pending confirmed Seller (manual confirm) — (already deducted)
confirmed shipped Seller —
shipped delivered Seller —
pending cancelled Seller (buyer ghosted) Restored
confirmed cancelled Seller Restored

13. Customer Auto-Creation from Checkout

When a buyer submits the checkout form, the system automatically creates or updates a customer record:

Buyer submits checkout → POST /api/checkout/[token]
    ↓
Extract: address_name, address_phone from submitted data
    ↓
Lookup customer by (shop_id, phone)
    ↓
├── EXISTS → update name if changed
└── NOT FOUND → INSERT new customer (name, phone, shop_id)
    ↓
Link customer_id to order → order.customer_id = customer.id
    ↓
Seller sees customer in /customers list with order history

Key rules:

Phone number is the unique identifier per shop (UNIQUE(shop_id, phone))
If phone already exists, we update the name but don't create a duplicate
This happens server-side in the checkout API route using the service-role client

15. Real-time Stock Updates (V1)

Live stock visibility is critical to prevent overselling. Use Supabase Realtime to push stock changes to all open dashboard sessions instantly.

15.1 How It Works

Seller A confirms an order → stock trigger deducts qty
    ↓
Supabase Realtime broadcasts UPDATE on product_variants
    ↓
Seller A's dashboard updates instantly (live counter)
Any other open tab/device also updates

15.2 Supabase Realtime Channel

// Conceptual — subscribe to variant stock changes for the seller's shop
const channel = supabase
  .channel('stock-updates')
  .on(
    'postgres_changes',
    {
      event: 'UPDATE',
      schema: 'public',
      table: 'product_variants',
      filter: `product_id=in.(${productIds})`,  // only this shop's products
    },
    (payload) => {
      // payload.new.stock_qty → update local state
      // If stock_qty <= low_stock_threshold → show toast warning
    }
  )
  .subscribe();

15.3 Where Real-time Updates Appear

Page What updates
/dashboard Low-stock alerts counter, "stock holding" count
/products Stock qty badge per product refreshes live
/products/[id] Variant stock numbers update in real-time
/orders/new Available stock shown when building cart
